# --------------------------------------------------------------
# Helper Functions
# --------------------------------------------------------------
def find_column(df, col_name):
    """Case-insensitive column lookup"""
    for c in df.columns:
        if c.lower() == col_name.lower():
            return c
    return None
def get_existing_columns(df, columns):
    """Return list of columns that exist in the DataFrame"""
    existing = []
    for c in columns:
        found = find_column(df, c)
        if found:
            existing.append(found)
    return existing
def write_to_s3(df, s3_path, mode="overwrite"):
    """Write DataFrame to S3 as Parquet"""
    full_path = f"s3://{raw_vault_bucket}/{s3_path}"
    df.write.mode(mode).parquet(full_path)
    logger.info("Written %d rows to %s", df.count(), full_path)
def read_bronze_table(table_name):
    """Read a table from bronze database"""
    try:
        df = spark.table(f"{bronze_db}.{table_name.lower()}")
        logger.info("Read table %s with %d columns", table_name, len(df.columns))
        return df
    except Exception as e:
        logger.warning("Could not read table %s: %s", table_name, e)
        return None
def generate_hash_key(df, columns, key_name="hash_key"):
    """Generate SHA-256 hash key from multiple columns"""
    existing_cols = get_existing_columns(df, columns)
    if not existing_cols:
        return None
    concat_expr = concat_ws("||", *[coalesce(col(c).cast(StringType()), lit("")).alias(c) for c in existing_cols])
    return df.withColumn(key_name, sha2(concat_expr, 256))
# --------------------------------------------------------------
# Create HUB Tables
# --------------------------------------------------------------
def create_hubs():
    logger.info("=" * 60)
    logger.info("Creating HUB Tables")
    logger.info("=" * 60)
    
    hub_dataframes = {}
    
    for hub in VAULT_MODEL["hubs"]:
        hub_name = hub["hub_name"]
        bk_columns = hub["business_key_columns"]
        source_tables = hub["source_tables"]
        
        logger.info("Building HUB: %s from sources: %s", hub_name, source_tables)
        
        hub_dfs = []
        for src_table in source_tables:
            src_df = read_bronze_table(src_table)
            if src_df is None:
                continue
            
            existing_bk_cols = get_existing_columns(src_df, bk_columns)
            if not existing_bk_cols:
                logger.warning("No business key columns found in %s for %s", src_table, hub_name)
                continue
            
            # Select business key columns and add metadata
            select_cols = [col(c).alias(c.lower()) for c in existing_bk_cols]
            key_df = src_df.select(*select_cols).distinct()
            
            # Generate hash key
            concat_expr = concat_ws("||", *[coalesce(col(c.lower()).cast(StringType()), lit("")) for c in existing_bk_cols])
            key_df = key_df.withColumn("hub_hash_key", sha2(concat_expr, 256))
            key_df = key_df.withColumn("load_date", current_timestamp())
            key_df = key_df.withColumn("record_source", lit(src_table))
            
            # Create business_key as concatenated string for reference
            key_df = key_df.withColumn("business_key", concat_ws("|", *[col(c.lower()) for c in existing_bk_cols]))
            
            hub_dfs.append(key_df)
        
        if not hub_dfs:
            logger.warning("No data for HUB %s", hub_name)
            continue
        
        # Union all sources
        hub_df = hub_dfs[0]
        for df in hub_dfs[1:]:
            hub_df = hub_df.unionByName(df, allowMissingColumns=True)
        
        hub_df = hub_df.dropDuplicates(["hub_hash_key"])
        
        # Write to S3
        write_to_s3(hub_df, f"{hub_path}/{hub_name}")
        hub_dataframes[hub_name] = f"s3://{raw_vault_bucket}/{hub_path}/{hub_name}"
    
    return hub_dataframes

# Create SATELLITE Tables
# --------------------------------------------------------------
def create_satellites(hub_paths):
    logger.info("=" * 60)
    logger.info("Creating SATELLITE Tables")
    logger.info("=" * 60)
    
    for sat in VAULT_MODEL["satellites"]:
        sat_name = sat["sat_name"]
        hub_ref = sat["hub_reference"]
        hub_key_columns = sat["hub_key_columns"]
        attributes = sat["attributes"]
        source_table = sat["source_table"]
        
        logger.info("Building SATELLITE: %s from source: %s", sat_name, source_table)
        
        # Read source table
        src_df = read_bronze_table(source_table)
        if src_df is None:
            logger.warning("Could not read source table %s for satellite %s", source_table, sat_name)
            continue
        
        # Get existing hub key columns
        existing_hub_keys = get_existing_columns(src_df, hub_key_columns)
        if not existing_hub_keys:
            logger.warning("No hub key columns found in %s for %s", source_table, sat_name)
            continue
        
        # Get existing attribute columns
        existing_attrs = get_existing_columns(src_df, attributes)
        if not existing_attrs:
            logger.warning("No attribute columns found in %s for %s", source_table, sat_name)
            continue
        
        # Generate hub hash key
        concat_expr = concat_ws("||", *[coalesce(col(c).cast(StringType()), lit("")) for c in existing_hub_keys])
        sat_df = src_df.withColumn("hub_hash_key", sha2(concat_expr, 256))
        
        # Generate hash diff for change detection
        attr_concat = concat_ws("||", *[coalesce(col(c).cast(StringType()), lit("")) for c in existing_attrs])
        sat_df = sat_df.withColumn("hash_diff", sha2(attr_concat, 256))
        
        # Add metadata
        sat_df = sat_df.withColumn("load_date", current_timestamp())
        sat_df = sat_df.withColumn("record_source", lit(source_table))
        
        # Select final columns
        select_cols = ["hub_hash_key"] + [col(c).alias(c.lower()) for c in existing_attrs] + ["hash_diff", "load_date", "record_source"]
        sat_df = sat_df.select(*select_cols).distinct()
        
        # Write to S3
        write_to_s3(sat_df, f"{sat_path}/{sat_name}")
# --------------------------------------------------------------
# Create LINK Tables
# --------------------------------------------------------------
def create_links(hub_paths):
    logger.info("=" * 60)
    logger.info("Creating LINK Tables")
    logger.info("=" * 60)
    
    for link in VAULT_MODEL["links"]:
        link_name = link["link_name"]
        hub_references = link["hub_references"]
        source_tables = link["source_tables"]
        
        logger.info("Building LINK: %s from sources: %s", link_name, source_tables)
        
        link_dfs = []
        for src_table in source_tables:
            src_df = read_bronze_table(src_table)
            if src_df is None:
                continue
            
            # Generate hash keys for each hub reference
            current_df = src_df
            hub_hash_keys = []
            valid = True
            
            for i, ref in enumerate(hub_references):
                hub_name = ref["hub_name"]
                ref_columns = ref["columns"]
                
                existing_cols = get_existing_columns(current_df, ref_columns)
                if not existing_cols:
                    logger.warning("Columns %s not found in %s for link %s", ref_columns, src_table, link_name)
                    valid = False
                    break
                
                # Generate hash key for this hub reference
                concat_expr = concat_ws("||", *[coalesce(col(c).cast(StringType()), lit("")) for c in existing_cols])
                hash_col_name = f"hk_{hub_name}"
                current_df = current_df.withColumn(hash_col_name, sha2(concat_expr, 256))
                hub_hash_keys.append(hash_col_name)
            
            if not valid:
                continue
            
            # Generate link hash key
            link_hash_expr = concat_ws("||", *[col(hk) for hk in hub_hash_keys])
            current_df = current_df.withColumn("link_hash_key", sha2(link_hash_expr, 256))
            
            # Add metadata
            current_df = current_df.withColumn("load_date", current_timestamp())
            current_df = current_df.withColumn("record_source", lit(src_table))
            
            # Select final columns
            select_cols = ["link_hash_key"] + hub_hash_keys + ["load_date", "record_source"]
            current_df = current_df.select(*select_cols).distinct()
            
            link_dfs.append(current_df)
        
        if not link_dfs:
            logger.warning("No data for LINK %s", link_name)
            continue
        
        # Union all sources
        link_df = link_dfs[0]
        for df in link_dfs[1:]:
            link_df = link_df.unionByName(df, allowMissingColumns=True)
        
        link_df = link_df.dropDuplicates(["link_hash_key"])
        
        # Write to S3
        write_to_s3(link_df, f"{link_path}/{link_name}")
# --------------------------------------------------------------
# Main Execution
# --------------------------------------------------------------
def main():
    logger.info("Starting Silver Layer (Raw Vault) Job")
    logger.info("=" * 60)
    
    # Create HUBs first
    hub_paths = create_hubs()
    
    # Create SATELLITEs
    create_satellites(hub_paths)
    
    # Create LINKs
    create_links(hub_paths)
    
    logger.info("=" * 60)
    logger.info("Silver Layer (Raw Vault) Job Completed Successfully")
if __name__ == "__main__":
    main()
    job.commit()
