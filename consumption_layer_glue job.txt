# Helper Functions
# --------------------------------------------------------------
def read_gold_table(table_name):
    """Read a table from gold layer"""
    path = f"{gold_path}/{table_name}"
    try:
        df = spark.read.parquet(path)
        logger.info("Read gold table %s with %d rows", table_name, df.count())
        return df
    except Exception as e:
        logger.warning("Could not read gold table %s: %s", table_name, e)
        return None

def write_consumption_view(df, view_name, mode="overwrite"):
    """Write DataFrame to consumption layer and register in Glue Catalog"""
    path = f"{consumption_path}/{view_name}"
    df.write.mode(mode).parquet(path)
    logger.info("Written %d rows to %s", df.count(), path)
    
    # Register table in Glue Catalog
    register_glue_table(view_name, path, df.schema)

def ensure_database_exists():
    """Create the Glue database if it doesn't exist"""
    try:
        glue_client.get_database(Name=glue_database)
        logger.info("Database %s already exists", glue_database)
    except glue_client.exceptions.EntityNotFoundException:
        logger.info("Creating database %s", glue_database)
        glue_client.create_database(
            DatabaseInput={
                'Name': glue_database,
                'Description': 'ProcureIQ Spend Consumption Layer Views'
            }
        )

def register_glue_table(table_name, s3_path, schema):
    """Register or update a table in the Glue Data Catalog"""
    columns = []
    for field in schema.fields:
        glue_type = spark_type_to_glue_type(field.dataType)
        columns.append({
            'Name': field.name,
            'Type': glue_type
        })
    
    table_input = {
        'Name': table_name,
        'Description': f'Consumption view: {table_name}',
        'StorageDescriptor': {
            'Columns': columns,
            'Location': s3_path,
            'InputFormat': 'org.apache.hadoop.hive.ql.io.parquet.MapredParquetInputFormat',
            'OutputFormat': 'org.apache.hadoop.hive.ql.io.parquet.MapredParquetOutputFormat',
            'SerdeInfo': {
                'SerializationLibrary': 'org.apache.hadoop.hive.ql.io.parquet.serde.ParquetHiveSerDe',
                'Parameters': {'serialization.format': '1'}
            },
            'Compressed': True
        },
        'TableType': 'EXTERNAL_TABLE',
        'Parameters': {
            'classification': 'parquet',
            'compressionType': 'snappy'
        }
    }
    
    try:
        glue_client.get_table(DatabaseName=glue_database, Name=table_name)
        # Table exists, update it
        glue_client.update_table(
            DatabaseName=glue_database,
            TableInput=table_input
        )
        logger.info("Updated Glue table: %s.%s", glue_database, table_name)
    except glue_client.exceptions.EntityNotFoundException:
        # Table doesn't exist, create it
        glue_client.create_table(
            DatabaseName=glue_database,
            TableInput=table_input
        )
        logger.info("Created Glue table: %s.%s", glue_database, table_name)

def spark_type_to_glue_type(spark_type):
    """Convert Spark data type to Glue/Hive data type"""
    type_name = str(spark_type)
    
    if 'DecimalType' in type_name:
        # Extract precision and scale from DecimalType(p,s)
        import re
        match = re.search(r'DecimalType\((\d+),\s*(\d+)\)', type_name)
        if match:
            return f"decimal({match.group(1)},{match.group(2)})"
        return "decimal(10,2)"
    elif 'StringType' in type_name:
        return 'string'
    elif 'IntegerType' in type_name:
        return 'int'
    elif 'LongType' in type_name:
        return 'bigint'
    elif 'DoubleType' in type_name:
        return 'double'
    elif 'FloatType' in type_name:
        return 'float'
    elif 'DateType' in type_name:
        return 'date'
    elif 'TimestampType' in type_name:
        return 'timestamp'
    elif 'BooleanType' in type_name:
        return 'boolean'
    else:
        return 'string'

def get_available_columns(df, requested_columns):
    """Get list of columns that exist in the DataFrame"""
    df_cols_lower = {c.lower(): c for c in df.columns}
    available = []
    for col_name in requested_columns:
        if col_name.lower() in df_cols_lower:
            available.append(df_cols_lower[col_name.lower()])
    return available

# Create Standard Views
# --------------------------------------------------------------
def create_standard_views():
    """Create all standard views from gold tables"""
    logger.info("Creating standard consumption views...")
    
    for view_name, view_def in VIEW_DEFINITIONS.items():
        source_table = view_def["source_table"]
        requested_columns = view_def["columns"]
        
        logger.info("Creating view: %s from source: %s", view_name, source_table)
        
        # Read source table
        df = read_gold_table(source_table)
        if df is None:
            logger.warning("Skipping view %s - source table %s not found", view_name, source_table)
            continue
        
        # Select available columns
        available_cols = get_available_columns(df, requested_columns)
        if not available_cols:
            logger.warning("Skipping view %s - no matching columns found", view_name)
            continue
        
        # Create view DataFrame
        view_df = df.select([col(c) for c in available_cols])
        
        # Rename columns to lowercase to match view definition
        for old_col, new_col in zip(available_cols, [c.lower() for c in available_cols]):
            if old_col != new_col:
                view_df = view_df.withColumnRenamed(old_col, new_col)
        
        # Write view
        write_consumption_view(view_df, view_name)

def create_fact_all_sources_view():
    """Create fact_all_sources_vw with special transformation"""
    logger.info("Creating fact_all_sources_vw...")
    
    df = read_gold_table("invoice_status_history")
    if df is None:
        logger.warning("Cannot create fact_all_sources_vw - source table not found")
        return
    
    # Apply ROW_NUMBER to get latest status per invoice
    window_spec = Window.partitionBy("invoice_number", "company_code", "fiscal_year") \
                        .orderBy(col("sequence_nbr").desc())
    
    df = df.withColumn("row_num", row_number().over(window_spec))
    df = df.filter(col("row_num") == 1).drop("row_num")
    
    # Select and rename columns for fact_all_sources_vw
    result = df.select(
        col("invoice_number").alias("document_number"),
        col("invoice_number"),
        col("company_code"),
        col("plant_code"),
        col("fiscal_year"),
        col("posting_date"),
        col("document_date"),
        col("vendor_id"),
        col("currency"),
        col("invoice_amount_local"),
        col("invoice_amount_local").alias("invoice_amount_document"),
        col("gl_account"),
        col("baseline_date"),
        col("payment_terms"),
        col("status").alias("invoice_status"),
        col("status_notes"),
        col("payment_date"),
        col("clearing_document"),
        col("aging_days"),
        col("purchase_order_reference"),
        col("po_amount"),
        col("po_purpose"),
        col("document_type"),
        col("accrual_indicator"),
        col("due_date"),
        col("invoice_receipt_date"),
        col("discount_percent"),
        col("discount_days"),
        col("net_due_days"),
        col("region"),
        col("system")
    )
    
    write_consumption_view(result, "fact_all_sources_vw")

    logger.info("Starting Consumption Layer Job")
