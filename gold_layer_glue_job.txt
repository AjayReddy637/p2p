def read_silver_table(table_type, table_name):
    """Read a table from silver layer"""
    if table_type == "hub":
        path = f"{silver_hub_path}/{table_name}"
    elif table_type == "link":
        path = f"{silver_link_path}/{table_name}"
    elif table_type == "sat":
        path = f"{silver_sat_path}/{table_name}"
    else:
        raise ValueError(f"Unknown table type: {table_type}")
    
    try:
        df = spark.read.parquet(path)
        logger.info("Read %s table %s with %d rows", table_type, table_name, df.count())
        return df
    except Exception as e:
        logger.warning("Could not read %s table %s: %s", table_type, table_name, e)
        return None
def write_gold_table(df, table_name, mode="overwrite"):
    """Write DataFrame to gold layer"""
    path = f"{gold_path}/{table_name}"
    df.write.mode(mode).parquet(path)
    logger.info("Written %d rows to %s", df.count(), path)
def safe_col(df, col_name, default=None):
    """Safely reference a column with a default value if it doesn't exist"""
    if col_name.lower() in [c.lower() for c in df.columns]:
        return col(col_name)
    return lit(default)

def create_cash_flow_forecast():
    """Create cash_flow_forecast table"""
    logger.info("Creating cash_flow_forecast")
    
    sat_invoice = read_silver_table("sat", "SAT_INVOICE_RBKP")
    
    if sat_invoice is None:
        logger.warning("Cannot create cash_flow_forecast - missing source tables")
        return
    
    # Calculate days until due and bucket
    df = sat_invoice.withColumn(
        "days_until_due",
        datediff(col("zfbdt"), current_date())
    ).withColumn(
        "forecast_bucket",
        when(col("days_until_due") < 0, "Overdue")
        .when(col("days_until_due") <= 7, "Due in 7 days")
        .when(col("days_until_due") <= 30, "Due in 30 days")
        .when(col("days_until_due") <= 60, "Due in 60 days")
        .when(col("days_until_due") <= 90, "Due in 90 days")
        .otherwise("Due beyond 90 days")
    )
    
    result = df.groupBy("forecast_bucket").agg(
        count("*").cast(DecimalType(10, 6)).alias("invoice_count"),
        spark_sum(coalesce(col("rmwwr"), lit(0))).cast(DecimalType(10, 2)).alias("total_amount"),
        spark_min(col("zfbdt")).cast(DateType()).alias("earliest_due"),
        spark_max(col("zfbdt")).cast(DateType()).alias("latest_due")
    )
    
    write_gold_table(result, "cash_flow_forecast")

        spark_sum(col("late_amount")).cast(DecimalType(9, 2)).alias("late_payment_amount"),

        spark_sum(col("dmbtr")).cast(DecimalType(9, 2)).alias("total_payment_amount")
    ).withColumn(
        "late_payment_rate_pct",
        (col("late_payment_amount") / col("total_payment_amount")).cast(DecimalType(11, 8))
    ).select(
        col("year").cast(DecimalType(4, 0)),
        col("month").cast(DecimalType(2, 0)),
        col("late_payment_amount"),
        col("late_payment_count"),
        col("total_payment_amount"),
        col("late_payment_rate_pct")
    )
    
    write_gold_table(result, "late_payment_amount")

def create_duplicate_payments_for_invoice():
    """Create duplicate_payments_for_invoice table"""
    logger.info("Creating duplicate_payments_for_invoice")
    
    sat_cleared = read_silver_table("sat", "SAT_CLEARED_ITEM_BSAK")
    
    if sat_cleared is None:
        logger.warning("Cannot create duplicate_payments_for_invoice - missing source tables")
        return
    
    # Group by invoice and count payments
    df = sat_cleared.withColumn("year", year(col("budat"))) \
                    .withColumn("month", month(col("budat")))
    
    window = Window.partitionBy("belnr", "bukrs", "gjahr")
    
    df = df.withColumn("payment_count", count("*").over(window))
    
    # Filter to potential duplicates (more than 1 payment per invoice)
    duplicates = df.filter(col("payment_count") > 1)
    
    result = duplicates.groupBy("belnr", "year", "month").agg(
        spark_sum(col("dmbtr")).cast(DecimalType(7, 2)).alias("duplicate_payment_amount"),
        spark_sum(col("dmbtr")).cast(DecimalType(7, 2)).alias("total_payments")
    ).withColumn(
        "duplicate_payment_rate",
        (col("duplicate_payment_amount") / col("total_payments")).cast(DecimalType(11, 8))
    ).select(
        col("belnr").cast(DecimalType(7, 0)).alias("invoice_number"),
        col("year").cast(DecimalType(4, 0)),
        col("month").cast(DecimalType(2, 0)),
        col("duplicate_payment_amount"),
        col("total_payments"),
        col("duplicate_payment_rate")
    )
    
    write_gold_table(result, "duplicate_payments_for_invoice")
